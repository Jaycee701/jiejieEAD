# ============================================================================

# ----------------------------------------------------------------------------
#  安装：pip install -r requirements.txt
#  说明：全部使用成熟密码库，不手写底层算法，规避密码实现漏洞。
#        版本号采用「下限约束 + 兼容上限」，兼容 Python 3.8 ~ 3.11
#        （打包脚本内置的是 Python 3.11 embeddable）。
# ============================================================================

# ---------------- 模块一 & 模块二：对称加密 ----------------
pycryptodome>=3.15,<4.0        # AES-256-CBC 等标准算法（审计级实现）
gmssl>=3.2.1,<4.0              # 国密算法库：SM2 / SM3 / SM4

# ---------------- 模块二：中间人代理 ----------------
# 注意：mitmproxy 11+ 的 Web 界面访问需要 URL 中的 token，
#       本插件已在 GUI 侧解析 mitmweb 输出的带 token 地址（见 Main.vue 的 appendMitmLog）。
#       mitmproxy 12.x 需要 Python 3.12+；Python 3.11 环境会自动解析到 11.x。
mitmproxy>=9.0,<13.0           # 中间人代理框架（提供 mitmproxy/mitmweb 工具）

# ---------------- 模块三：国密分析与流量解析 ----------------
scapy>=2.5.0,<3.0              # 报文解析（读取 pcap，提取 TCP 载荷）
pyasn1>=0.4.8,<1.0             # ASN.1/DER 结构解析（公钥 PEM 结构校验）

# ---------------- 模块七：MCP 服务器（GUI「MCP 接入」页签依赖） ----------------
# 把加解密 / 文件扫描 / 能力矩阵暴露成 MCP 工具，供 WorkBuddy / Claude Desktop 调用。
# 下面两行 pin 不能省：mcp 默认会把 h11 升到 0.16、httpcore 升到 1.0.9，
# 而 mitmproxy 11.0.2 要求 h11<=0.14.0 —— 不压回去模块二 MITM 会起不来。
# h11==0.14.0 + httpcore==1.0.7 是实测 `pip check` 全绿、两侧共存的组合。
mcp>=1,<2                      # MCP SDK（FastMCP + stdio 传输）
h11==0.14.0                    # 与 mitmproxy 11.x 兼容（勿升到 0.16）
httpcore==1.0.7                # 与 h11 0.14 兼容（勿升到 1.0.9）

# ----------------------------------------------------------------------------
#  备注：
#  1. 若仅使用 CLI 加解密（模块一），只需 pycryptodome + gmssl；
#  2. 若仅使用国密分析中的 sm3hash / sm2check 子命令，无需 scapy；
#     scapy 仅 tlcp 子命令使用（本工具已做惰性导入，避免拖慢启动）；
#  3. mitmproxy 体积较大（含其 Web 界面依赖），仅模块二需要；
#  4. mcp 约 20MB，仅模块七（MCP 接入页签 + 外部客户端调用）需要。
# ----------------------------------------------------------------------------
